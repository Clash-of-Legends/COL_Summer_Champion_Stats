window.onload = async function () { const pickBanData = await loadXlsx('League_stats.xlsx', 'Pick-Ban'); const matchData = await loadXlsx('Stats.xlsx', 'Sheet1'); parseChampStats(pickBanData); parseMatchData(matchData); document.getElementById('sortSelect').addEventListener('change', () => renderGrid(document.getElementById('sortSelect').value) ); renderGrid('Picks'); }; let champs = [], matchRows = []; function loadXlsx(filename, sheetName) { return new Promise(resolve => { const reader = new FileReader(); reader.onload = function (e) { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, { type: 'array' }); const worksheet = workbook.Sheets[sheetName]; const json = XLSX.utils.sheet_to_json(worksheet, { header: 1 }); resolve(json); }; fetch(filename) .then(res => res.blob()) .then(blob => reader.readAsArrayBuffer(blob)); }); } function toStr(val) { return (val === undefined || val === null) ? '' : String(val).trim(); } const imageFixes = { 'Renata Glasc': 'Renata', 'Tahm Kench': 'TahmKench', 'Nunu & Willump': 'Nunu', "Jarvan IV": "JarvanIV", "Kai'Sa": "Kaisa", "Cho'Gath": "Chogath", "Vel'Koz": "Velkoz", "Bel'Veth": "Belveth", "Kha'Zix": "Khazix", "LeBlanc": "Leblanc", "Wukong": "MonkeyKing", "Nunu and Willump": "Nunu", "Dr. Mundo": "DrMundo" }; function parseChampStats(rows) { const gamesLine = rows[0]; const totalGames = toStr(gamesLine[1]) || '?'; const gameCountElem = document.getElementById('games'); if (gameCountElem) gameCountElem.textContent = `Počet her: ${totalGames}`; const headersRowIndex = rows.findIndex(r => toStr(r[1]) === 'Champ'); const headers = rows[headersRowIndex].slice(1); const dataRows = rows.slice(headersRowIndex + 1); champs = []; for (const row of dataRows) { const cols = row.slice(1); if (!cols[0]) continue; const champ = Object.fromEntries(headers.map((h, i) => [h, toStr(cols[i])])) const champName = champ['Champ']; const picks = parseInt(champ['Picks'] || '0'); const bans = parseInt(champ['Bans'] || '0'); if (!champName || champName.toLowerCase() === 'undefined' || (picks === 0 && bans === 0)) continue; champ['Wins'] = champ['Wins'] === 'undefined' || champ['Wins'] === '' ? '0' : champ['Wins']; champ['Loses'] = champ['Loses'] === 'undefined' || champ['Loses'] === '' ? '0' : champ['Loses']; champ['Winrate'] = formatPercent100(champ['Winrate']); champ['Banrate'] = formatPercent100(champ['Banrate']); champs.push(champ); } } function formatPercent100(val) { if (!val || val.includes('undefined')) return '0%'; const num = parseFloat(val.toString().replace('%', '').replace(',', '.')) || 0; return Math.round(num) + '%'; } function parseMatchData(rows) { const header = rows[0]; matchRows = rows.slice(1).map(row => { const rowMap = {}; header.forEach((key, i) => rowMap[toStr(key)] = toStr(row[i])); return { champ: rowMap['Champové'], player: rowMap['Hráč'], team: rowMap['Tým'], enemy: rowMap['Enemy'], result: rowMap['Win/Loss'], k: rowMap['Kills'], d: rowMap['Deaths'], a: rowMap['Assists'] }; }); } function renderGrid(sortBy) { const grid = document.getElementById('grid'); grid.innerHTML = ''; const sortValue = val => parseFloat((val || '0').toString().replace('%', '').replace(',', '.')) || 0; champs.sort((a, b) => sortValue(b[sortBy]) - sortValue(a[sortBy])); for (const champ of champs) { const champName = champ['Champ']; if (!champName || champName.toLowerCase() === 'undefined') continue; const fixedName = imageFixes[champName] || champName.replace(/[^a-zA-Z0-9]/g, ''); const imageUrl = `images/${fixedName}.jpg`; const card = document.createElement('div'); card.className = 'champ-card'; card.innerHTML = `
${champName} Winrate: ${champ['Winrate']}
Picks: ${champ['Picks']} | Bans: ${champ['Bans']}
Wins: ${champ['Wins']} | Loses: ${champ['Loses']}
Banrate: ${champ['Banrate']}
`; card.onclick = () => openModal(champName); grid.appendChild(card); } } function openModal(champName) { const modal = document.getElementById('modal'); const title = document.getElementById('modal-title'); const tbody = document.getElementById('modal-table').querySelector('tbody'); title.textContent = `Zápasy pro: ${champName}`; tbody.innerHTML = ''; const filtered = matchRows.filter(m => m.champ.trim() === champName.trim()); if (filtered.length === 0) { tbody.innerHTML = `Žádná data`; } else { for (const row of filtered) { const tr = document.createElement('tr'); const resultColor = row.result === 'Win' ? 'limegreen' : 'crimson'; tr.innerHTML = ` ${row.player} ${row.team} ${row.enemy} ${row.result} ${row.k} ${row.d} ${row.a} `; tbody.appendChild(tr); } } modal.style.display = 'block'; } function closeModal() { document.getElementById('modal').style.display = 'none'; }
